class HeroSection extends StatefulWidget {
  final VoidCallback onViewWork;
  const HeroSection({super.key, required this.onViewWork});

  @override
  State<HeroSection> createState() => _HeroSectionState();
}

class _HeroSectionState extends State<HeroSection> with TickerProviderStateMixin {
  late final AnimationController _ambient =
      AnimationController(vsync: this, duration: const Duration(seconds: 14))..repeat();
  late final AnimationController _shine =
      AnimationController(vsync: this, duration: const Duration(milliseconds: 3200))..repeat();
  late final AnimationController _intro =
      AnimationController(vsync: this, duration: const Duration(milliseconds: 1600))..forward();

  static const _roles = [
    'Mobile App Developer',
    'Cross-Platform Apps for Android & iOS',
    'Clean Architecture & GetX',
    'REST APIs & Firebase',
  ];
  int _roleIndex = 0;
  Timer? _roleTimer;

  @override
  void initState() {
    super.initState();
    _roleTimer = Timer.periodic(const Duration(milliseconds: 2600), (_) {
      if (mounted) setState(() => _roleIndex = (_roleIndex + 1) % _roles.length);
    });
  }

  @override
  void dispose() {
    _roleTimer?.cancel();
    _ambient.dispose();
    _shine.dispose();
    _intro.dispose();
    super.dispose();
  }

  /// Staggered intro: each element fades/slides in after [start] (0..1 of the intro).
  Widget _stagger(double start, Widget child) {
    final anim = CurvedAnimation(parent: _intro, curve: Interval(start, math.min(start + 0.45, 1.0), curve: Curves.easeOutCubic));
    return AnimatedBuilder(
      animation: anim,
      builder: (_, c) => Opacity(
        opacity: anim.value,
        child: Transform.translate(offset: Offset(0, 30 * (1 - anim.value)), child: c),
      ),
      child: child,
    );
  }

  @override
  Widget build(BuildContext context) {
    final mobile = isMobile(context);
    final size = MediaQuery.sizeOf(context);
    final align = mobile ? CrossAxisAlignment.center : CrossAxisAlignment.start;
    final textAlign = mobile ? TextAlign.center : TextAlign.start;

    final text = Column(
      crossAxisAlignment: align,
      children: [
        _stagger(0.0, const _AvailableBadge()),
        const SizedBox(height: 30),
        _stagger(
          0.1,
          Text.rich(
            TextSpan(children: [
              TextSpan(text: 'Hi, I\'m ', style: AppText.body(mobile ? 17 : 19, color: AppColors.textMuted)),
              TextSpan(
                text: PortfolioData.fullName,
                style: AppText.display(mobile ? 21 : 25, color: AppColors.text, weight: FontWeight.w600, height: 1.4),
              ),
            ]),
            textAlign: textAlign,
          ),
        ),
        const SizedBox(height: 10),
        _stagger(
          0.18,
          AnimatedBuilder(
            animation: _shine,
            builder: (_, child) => ShaderMask(
              blendMode: BlendMode.srcIn,
              shaderCallback: (rect) => LinearGradient(
                colors: const [AppColors.gold, AppColors.gold, AppColors.goldLight, AppColors.gold, AppColors.gold],
                stops: const [0, 0.4, 0.5, 0.6, 1],
                transform: _SlideGradient(_shine.value),
              ).createShader(rect),
              child: child,
            ),
            child: Text(
              PortfolioData.role,
              textAlign: textAlign,
              style: AppText.display(mobile ? 46 : 78, weight: FontWeight.w700, height: 1.05),
            ),
          ),
        ),
        const SizedBox(height: 16),
        _stagger(
          0.28,
          SizedBox(
            height: mobile ? 36 : 46,
            child: AnimatedSwitcher(
              duration: const Duration(milliseconds: 600),
              transitionBuilder: (child, anim) => FadeTransition(
                opacity: anim,
                child: SlideTransition(
                  position: Tween(begin: const Offset(0, 0.4), end: Offset.zero).animate(anim),
                  child: child,
                ),
              ),
              child: FittedBox(
                key: ValueKey(_roleIndex),
                fit: BoxFit.scaleDown,
                child: Text(
                  _roles[_roleIndex],
                  style: AppText.display(mobile ? 24 : 32, color: AppColors.goldLight, weight: FontWeight.w500)
                      .copyWith(fontStyle: FontStyle.italic),
                ),
              ),
            ),
          ),
        ),
        const SizedBox(height: 22),
        _stagger(
          0.36,
          ConstrainedBox(
            constraints: const BoxConstraints(maxWidth: 520),
            child: Text(PortfolioData.tagline,
                textAlign: textAlign, style: AppText.body(16.5, color: AppColors.textMuted, height: 1.8)),
          ),
        ),
        const SizedBox(height: 38),
        _stagger(
          0.46,
          Wrap(
            spacing: 16,
            runSpacing: 14,
            alignment: mobile ? WrapAlignment.center : WrapAlignment.start,
            children: [
              LuxButton(text: 'View My Work', icon: Icons.arrow_downward_rounded, onTap: widget.onViewWork),
              LuxButton(
                text: 'Download CV',
                icon: Icons.download_rounded,
                outlined: true,
                onTap: () => openLink(PortfolioData.cvPath),
              ),
            ],
          ),
        ),
        const SizedBox(height: 36),
        _stagger(
          0.56,
          Row(
            mainAxisSize: MainAxisSize.min,
            children: [
              _SocialIcon(icon: Icons.chat_rounded, tooltip: 'WhatsApp', url: PortfolioData.whatsapp, isWhatsApp: true),
              _SocialIcon(icon: Icons.code_rounded, tooltip: 'GitHub', url: PortfolioData.github),
              _SocialIcon(icon: Icons.work_outline_rounded, tooltip: 'LinkedIn', url: PortfolioData.linkedin),
              _SocialIcon(icon: Icons.mail_outline_rounded, tooltip: 'Email', url: 'mailto:${PortfolioData.email}'),
            ],
          ),
        ),
      ],
    );

    final portrait = _stagger(0.2, _Portrait(ambient: _ambient));

    return Stack(
      children: [
        // Slowly drifting golden glows in the background.
        Positioned.fill(
          child: AnimatedBuilder(
            animation: _ambient,
            builder: (_, __) {
              final t = _ambient.value * 2 * math.pi;
              return Stack(
                children: [
                  _Glow(
                    left: size.width * 0.62 + math.cos(t) * 60,
                    top: 60 + math.sin(t) * 40,
                    size: 520,
                    color: AppColors.powder.withValues(alpha: 0.55),
                  ),
                  _Glow(
                    left: -160 + math.sin(t) * 50,
                    top: 360 + math.cos(t) * 50,
                    size: 460,
                    color: const Color(0xFFF6D9C9).withValues(alpha: 0.6),
                  ),
                ],
              );
            },
          ),
        ),
        SectionContainer(
          child: Column(
            children: [
              SizedBox(height: mobile ? 10 : 30),
              mobile
                  ? Column(children: [FittedBox(fit: BoxFit.scaleDown, child: portrait), const SizedBox(height: 44), text])
                  : Row(
                      children: [
                        Expanded(flex: 6, child: text),
                        const SizedBox(width: 40),
                        Expanded(flex: 5, child: Center(child: FittedBox(fit: BoxFit.scaleDown, child: portrait))),
                      ],
                    ),
              SizedBox(height: mobile ? 56 : 90),
              _stagger(0.65, const _StatsRow()),
            ],
          ),
        ),
      ],
    );
  }
}

class _SlideGradient extends GradientTransform {
  final double percent;
  const _SlideGradient(this.percent);

  @override
  Matrix4? transform(Rect bounds, {TextDirection? textDirection}) =>
      Matrix4.translationValues(bounds.width * (percent * 2.4 - 1.2), 0, 0);
}

class _Glow extends StatelessWidget {
  final double left, top, size;
  final Color color;
  const _Glow({required this.left, required this.top, required this.size, required this.color});

  @override
  Widget build(BuildContext context) {
    return Positioned(
      left: left,
      top: top,
      child: IgnorePointer(
        child: Container(
          width: size,
          height: size,
          decoration: BoxDecoration(
            shape: BoxShape.circle,
            gradient: RadialGradient(colors: [color, color.withValues(alpha: 0)]),
          ),
        ),
      ),
    );
  }
}

class _SocialIcon extends StatefulWidget {
  final IconData icon;
  final String tooltip;
  final String url;
  final bool isWhatsApp;
  const _SocialIcon({required this.icon, required this.tooltip, required this.url, this.isWhatsApp = false});

  @override
  State<_SocialIcon> createState() => _SocialIconState();
}

class _SocialIconState extends State<_SocialIcon> {
  bool _hover = false;

  @override
  Widget build(BuildContext context) {
    return Padding(
      padding: const EdgeInsets.symmetric(horizontal: 6),
      child: Tooltip(
        message: widget.tooltip,
        child: MouseRegion(
          cursor: SystemMouseCursors.click,
          onEnter: (_) => setState(() => _hover = true),
          onExit: (_) => setState(() => _hover = false),
          child: GestureDetector(
            onTap: () => openLink(widget.url),
            child: AnimatedContainer(
              duration: const Duration(milliseconds: 250),
              width: 46,
              height: 46,
              decoration: BoxDecoration(
                shape: BoxShape.circle,
                color: _hover ? AppColors.gold : Colors.transparent,
                border: Border.all(color: _hover ? AppColors.gold : AppColors.border),
              ),
              child: widget.isWhatsApp
                  ? Center(child: WhatsAppIcon(size: 21, color: _hover ? AppColors.background : AppColors.textMuted))
                  : Icon(widget.icon, size: 20, color: _hover ? AppColors.background : AppColors.textMuted),
            ),
          ),
        ),
      ),
    );
  }
}

/// Arch-shaped portrait with an offset gold frame and floating badges.
class _Portrait extends StatelessWidget {
  final AnimationController ambient;
  const _Portrait({required this.ambient});

  @override
  Widget build(BuildContext context) {
    const w = 340.0, h = 430.0;
    return SizedBox(
      width: w + 300,
      height: h + 100,
      child: AnimatedBuilder(
        animation: ambient,
        builder: (_, __) {
          final t = ambient.value * 2 * math.pi;
          double float(double phase) => math.sin(t * 2 + phase) * 8;
          return Stack(
            clipBehavior: Clip.none,
            children: [
              // Offset gold outline frame.
              Positioned(
                left: 174,
                top: 30,
                child: Container(
                  width: w,
                  height: h,
                  decoration: BoxDecoration(
                    borderRadius: const BorderRadius.vertical(top: Radius.circular(w / 2), bottom: Radius.circular(24)),
                    border: Border.all(color: AppColors.gold.withValues(alpha: 0.55), width: 1.2),
                  ),
                ),
              ),
              // Photo.
              Positioned(
                left: 150,
                top: 6,
                child: Container(
                  width: w,
                  height: h,
                  decoration: BoxDecoration(
                    borderRadius: const BorderRadius.vertical(top: Radius.circular(w / 2), bottom: Radius.circular(24)),
                    gradient: const LinearGradient(
                      colors: [AppColors.surfaceLight, AppColors.surface],
                      begin: Alignment.topCenter,
                      end: Alignment.bottomCenter,
                    ),
                    boxShadow: [
                      BoxShadow(color: AppColors.shadow, blurRadius: 50, offset: const Offset(0, 30)),
                    ],
                  ),
                  child: ClipRRect(
                    borderRadius: const BorderRadius.vertical(top: Radius.circular(w / 2), bottom: Radius.circular(24)),
                    child: Image.asset(
                      'assets/images/profile.jpeg',
                      fit: BoxFit.cover,
                      errorBuilder: (_, __, ___) => Center(
                        child: Text('M', style: AppText.display(140, color: AppColors.gold, weight: FontWeight.w700)),
                      ),
                    ),
                  ),
                ),
              ),
              Positioned(
                left: 0,
                top: 0 + float(0),
                child: _FloatBadge(icon: Icons.phone_iphone_rounded, title: '${PortfolioData.projects.length}+ Apps', sub: 'Built & shipped'),
              ),
              Positioned(
                right: 0,
                top: 448 + float(1.6),
                child: const _FloatBadge(icon: Icons.auto_awesome_rounded, title: 'Clean Code', sub: 'GetX · Firebase'),
              ),
            ],
          );
        },
      ),
    );
  }
}

/// Animated counters.
class _StatsRow extends StatelessWidget {
  const _StatsRow();

  @override
  Widget build(BuildContext context) {
    final skillsCount = PortfolioData.skills.fold<int>(0, (sum, g) => sum + g.skills.length);
    final stats = [
      (PortfolioData.projects.length, '+', 'Apps Built'),
      (skillsCount, '+', 'Skills & Tools'),
      (PortfolioData.training.length, '', 'Courses & Trainings'),
    ];
    return Container(
      padding: const EdgeInsets.symmetric(vertical: 28, horizontal: 12),
      decoration: const BoxDecoration(
        border: Border(top: BorderSide(color: AppColors.border), bottom: BorderSide(color: AppColors.border)),
      ),
      child: Wrap(
        alignment: WrapAlignment.spaceAround,
        runSpacing: 24,
        spacing: 24,
        children: stats
            .map(
              (s) => SizedBox(
                width: 200,
                child: Column(
                  children: [
                    TweenAnimationBuilder<double>(
                      tween: Tween(begin: 0, end: s.$1.toDouble()),
                      duration: const Duration(milliseconds: 2200),
                      curve: Curves.easeOutCubic,
                      builder: (_, v, __) => Text('${v.round()}${s.$2}',
                          style: AppText.display(42, color: AppColors.gold, weight: FontWeight.w600, height: 1.1)),
                    ),
                    const SizedBox(height: 6),
                    Text(s.$3, style: AppText.body(13.5, color: AppColors.textMuted, letterSpacing: 0.5)),
                  ],
                ),
              ),
            )
            .toList(),
      ),
    );
  }
}
