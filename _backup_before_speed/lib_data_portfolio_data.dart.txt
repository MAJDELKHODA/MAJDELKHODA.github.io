class Project {
  final String title;
  final String subtitle;
  final String description;
  final List<String> tags;
  final List<String> images; // files in assets/images/ — first one shows on the card
  final Alignment imageAlignment; // which part of the image shows on the card
  final String link;
  final String linkLabel;
  final IconData linkIcon;
  final IconData icon;
  final Color color;
  final bool isTeam;
  final String? demoUrl; // optional video link (YouTube / Drive) → shows "Watch Demo"

  const Project({
    required this.title,
    required this.subtitle,
    required this.description,
    required this.tags,
    this.images = const [],
    this.imageAlignment = Alignment.center,
    required this.link,
    this.linkLabel = 'View on GitHub',
    this.linkIcon = Icons.code_rounded,
    required this.icon,
    required this.color,
    this.isTeam = false,
    this.demoUrl,
  });
}
