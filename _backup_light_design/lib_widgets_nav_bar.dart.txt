import 'package:flutter/material.dart';

import '../data/portfolio_data.dart';
import '../theme/app_colors.dart';
import 'common.dart';

class NavItem {
  final String label;
  final GlobalKey key;
  const NavItem(this.label, this.key);
}

class NavBar extends StatelessWidget {
  final List<NavItem> items;
  final void Function(GlobalKey key) onTap;
  const NavBar({super.key, required this.items, required this.onTap});

  @override
  Widget build(BuildContext context) {
    final mobile = isMobile(context);
    return Container(
      height: 72,
      padding: EdgeInsets.symmetric(horizontal: mobile ? 20 : 40),
      decoration: BoxDecoration(
        color: AppColors.background.withValues(alpha: 0.95),
        border: const Border(bottom: BorderSide(color: AppColors.border)),
      ),
      child: Center(
        child: ConstrainedBox(
          constraints: const BoxConstraints(maxWidth: 1100),
          child: Row(
            children: [
              const _Logo(),
              const Spacer(),
              if (mobile)
                Builder(
                  builder: (ctx) => IconButton(
                    icon: const Icon(Icons.menu_rounded, color: AppColors.text),
                    onPressed: () => Scaffold.of(ctx).openEndDrawer(),
                  ),
                )
              else
                ...items.map(
                  (i) => Padding(
                    padding: const EdgeInsets.only(left: 8),
                    child: TextButton(
                      onPressed: () => onTap(i.key),
                      style: TextButton.styleFrom(foregroundColor: AppColors.text),
                      child: Text(i.label, style: const TextStyle(fontWeight: FontWeight.w500)),
                    ),
                  ),
                ),
            ],
          ),
        ),
      ),
    );
  }
}

class _Logo extends StatelessWidget {
  const _Logo();

  @override
  Widget build(BuildContext context) {
    return Row(
      children: [
        Container(
          width: 38,
          height: 38,
          alignment: Alignment.center,
          decoration: BoxDecoration(
            gradient: AppColors.heroGradient,
            borderRadius: BorderRadius.circular(12),
          ),
          child: Text(
            PortfolioData.name[0],
            style: const TextStyle(color: Colors.white, fontWeight: FontWeight.w700, fontSize: 18),
          ),
        ),
        const SizedBox(width: 10),
        const Text(
          PortfolioData.name,
          style: TextStyle(fontWeight: FontWeight.w700, fontSize: 18, color: AppColors.text),
        ),
        const Text(
          '.',
          style: TextStyle(fontWeight: FontWeight.w700, fontSize: 22, color: AppColors.accent),
        ),
      ],
    );
  }
}

class NavDrawer extends StatelessWidget {
  final List<NavItem> items;
  final void Function(GlobalKey key) onTap;
  const NavDrawer({super.key, required this.items, required this.onTap});

  @override
  Widget build(BuildContext context) {
    return Drawer(
      backgroundColor: AppColors.background,
      child: SafeArea(
        child: ListView(
          padding: const EdgeInsets.all(20),
          children: [
            const _Logo(),
            const SizedBox(height: 24),
            ...items.map(
              (i) => ListTile(
                title: Text(i.label, style: const TextStyle(fontWeight: FontWeight.w500)),
                onTap: () {
                  Navigator.of(context).pop();
                  onTap(i.key);
                },
              ),
            ),
          ],
        ),
      ),
    );
  }
}
